$RootPath = "C:\Users\ELARBI\Desktop\medical-record-app\backend\src" # <- Mets ici ton chemin

$MaxDepth = 8                   # <- Nombre de niveaux à explorer

 function Get-DirectoryTree {

   param(

     [string]$Path,

     [int]$Depth

   )



   if ($Depth -lt 0) { return }



   # Liste les fichiers et dossiers dans le répertoire courant

   Get-ChildItem -LiteralPath $Path | ForEach-Object {

     $item = $_

     $indent = " " * (($MaxDepth - $Depth) * 4) # Indentation selon profondeur

     Write-Output "$indent- $($item.Name)"



     # Si c'est un dossier, appel récursif

     if ($item.PSIsContainer) {

       # 🔸 Important : utiliser FullName tel quel avec -LiteralPath pour éviter les problèmes de crochets

       Get-DirectoryTree -Path $item.FullName -Depth ($Depth - 1)

     }

   }

 }



Get-DirectoryTree -Path $RootPath -Depth $MaxDepth
